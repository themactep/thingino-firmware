# U-Env for Jooan S7-U (T23ZN, SC2336, ATBM6441)
serialport=ttyS1

# GPIO init: PTZ pan 46/45/39/38 (PB13/14 double as unused SDIO DAT2/3,
# board is 1-bit), tilt 53/52/57/58, ircut 50/49, IR LED 60, white light 59,
# speaker enable 63 (active high)
gpio_default=38o 39o 45o 46o 49o 50o 52o 53o 57o 58o 59o 60o 63o

# Makefile.utils expands init=${init} into bootargs, so this carries the
# built-in jzmmc driver's SD card-detect GPIO 61 on the kernel command line.
init=/init jzmmc_v12.cd_gpio_pin=61
